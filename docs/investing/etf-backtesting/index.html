# 0. Imports & global parameters
###############################################################################
import pandas as pd
import numpy as np
import yfinance as yf
from pathlib import Path
from datetime import datetime, timedelta

ETF_LIST = ["QQQ", "QLD", "VOO", "SSO"]
START      = "2007-01-01"        # enough history to cover post‑GFC regime
FEE_PCT    = 0.0003              # 0.03 %
CASH_TICK  = "BIL"               # 1‑3 M T‑Bill ETF as cash proxy (optional)

###############################################################################
# 1. Data loader with lightweight local caching
###############################################################################
def get_data(tickers, start=START, end=None, cache_dir="cache"):
    """
    Download Adj Close & Volume for tickers since `start`, return wide DataFrame.
    Cached locally (CSV) to avoid repeated network calls.
    """
    end = end or datetime.today().strftime("%Y-%m-%d")
    cache = Path(cache_dir)
    cache.mkdir(exist_ok=True)

    all_df = []
    for t in tickers:
        f = cache / f"{t}.csv"
        if f.exists():
            df = pd.read_csv(f, index_col=0, parse_dates=True)
            # refresh the last 14 days to incorporate splits/divs
            if (df.index[-1] < pd.to_datetime(end) - timedelta(days=2)):
                df_new = yf.download(t, start=df.index[-14], end=end, progress=False)[["Adj Close", "Volume"]]
                df = pd.concat([df[:-14], df_new])
                df.to_csv(f)
        else:
            df = yf.download(t, start=start, end=end, progress=False)[["Adj Close", "Volume"]]
            df.to_csv(f)
        df = df.rename(columns={"Adj Close": t, "Volume": t+"_VOL"})
        all_df.append(df)
    return pd.concat(all_df, axis=1)

###############################################################################
# 2. Feature engineering helpers
###############################################################################
def add_features(price_df):
    """
    Given wide Px DF (Adj Close), return a DF of engineered features:
    - 5‑day momentum (%)
    - 20‑day SMA ratio
    - 20‑day historical volatility (daily %)
    """
    feat = pd.DataFrame(index=price_df.index)
    for t in ETF_LIST:
        px = price_df[t]
        feat[f"{t}_mom5"]  = px.pct_change(5)
        feat[f"{t}_sma20"] = px / px.rolling(20).mean() - 1          # >0 => above SMA
        feat[f"{t}_vol20"] = np.log(px).diff().rolling(20).std() * np.sqrt(252)
    return feat

###############################################################################
# 3. Baseline *paired‑switch* strategy
###############################################################################
def baseline_weights(price_df, feat_df):
    """
    Rule: For each underlying/leveraged pair,
    • If 5‑day momentum of the underlying ETF > 0 ⇒ use its 2× leveraged sibling
    • Else ⇒ use the un‑levered ETF.
    We keep 50 % of the portfolio in the NASDAQ pair (QQQ/QLD)
    and 50 % in the S&P‑500 pair (VOO/SSO).
    Output: daily weights DataFrame (rows = dates, cols = ETFs).
    """
    w = pd.DataFrame(0.0, index=price_df.index, columns=ETF_LIST)

    # condition booleans
    use_qld = feat_df["QQQ_mom5"] > 0
    use_sso = feat_df["VOO_mom5"] > 0

    w.loc[use_qld, "QLD"] = 0.50
    w.loc[~use_qld, "QQQ"] = 0.50

    w.loc[use_sso, "SSO"] = 0.50
    w.loc[~use_sso, "VOO"] = 0.50

    return w

###############################################################################
# 4. Vectorised back‑tester (daily rebalancing, fee‑aware)
###############################################################################
def backtest(price_df, weight_df, fee_pct=FEE_PCT):
    """
    Assumes weights are *target* EOD allocations (fully invested).
    Implements next‑day‑open execution by shifting weights by 1.
    """
    rebal_dates = weight_df.index
    # shift so that today's weights trade at tomorrow's open/close
    w = weight_df.shift().fillna(0)

    # daily returns
    rets = price_df[ETF_LIST].pct_change().loc[w.index]
    # portfolio returns before cost
    port_ret_pre_cost = (w * rets).sum(1)

    # turnover: |Δw| / 2 (two‑sided) — efficient vectorised calc
    dw = w.diff().abs()
    daily_turnover = dw.sum(1)
    cost = daily_turnover * fee_pct

    port_ret_after_cost = port_ret_pre_cost - cost

    nav = (1 + port_ret_after_cost).cumprod().rename("NAV")
    out_df = pd.concat([nav, port_ret_after_cost.rename("ret"), daily_turnover.rename("turnover")], axis=1)
    return out_df

###############################################################################
# 5. Performance summary
###############################################################################
def perf_stats(nav_series):
    """
    Computes CAGR, annualised vol & Sharpe, max DD.
    """
    nav = nav_series
    daily_ret = nav.pct_change().dropna()
    yrs = (nav.index[-1] - nav.index[0]).days / 365.25
    cagr = nav.iloc[-1] ** (1/yrs) - 1
    vol  = daily_ret.std() * np.sqrt(252)
    sharpe = cagr / vol if vol > 0 else np.nan
    dd = (nav / nav.cummax() - 1).min()
    return pd.Series({"CAGR": cagr, "Vol": vol, "Sharpe": sharpe, "MaxDD": dd})

# 6. End‑to‑end run (executed when script is run directly)
###############################################################################
if __name__ == "__main__":
    price = get_data(ETF_LIST, START)
    feats = add_features(price[ETF_LIST])           # only Adj Close cols
    weights = baseline_weights(price, feats)
    bt = backtest(price, weights)
    stats = perf_stats(bt["NAV"])
    print("====== Baseline Back‑test Results ======")
    print(stats.apply(lambda x: f"{x:.2%}" if x<1 else f"{x:,.2f}"))

    What the script does

    1. Downloads & caches daily adjusted close and volume for the four ETFs (from 1 Jan 2007 to today).

    2. Creates features (5‑day momentum, 20‑day SMA ratio, 20‑day vol).

    3. Generates weights each evening:
      If QQQ’s short‑term momentum > 0 ⇒ hold QLD (2× levered).
      Otherwise hold QQQ.
      Similar logic for VOO/SSO.

    4. Executes the weights the next trading day at the close (simplest to code; switch to open by pulling intraday if desired).

    5. Subtracts a flat 0.03 % fee on every dollar of daily turnover.

    6. Outputs an equity curve and key performance stats.

# 1. Version‑proof Data loader (works for yfinance 0.2 and 0.3)
###############################################################################
def get_data(tickers, start=START, end=None, cache_dir="cache"):
    """
    Returns wide DataFrame with columns:
        <TICKER>         – adjusted close price (float)
        <TICKER>_VOL     – daily volume         (int)

    Robust to:
    • single vs. multiple tickers
    • MultiIndex columns in either order
    • yfinance auto‑adjust on/off
    """
    end = end or datetime.today().strftime("%Y-%m-%d")
    cache_path = Path(cache_dir)
    cache_path.mkdir(exist_ok=True)

    out_frames = []
    for t in tickers:

        # ---------------- download or refresh ----------------
        if csv_f.exists():
            df = pd.read_csv(csv_f, index_col=0, parse_dates=True)
            staleness = (pd.Timestamp(end) - df.index[-1]).days
            if staleness > 2:       # refresh last 2 weeks
                df_new = yf.download(
                    t, start=df.index[-14], end=end,
                    auto_adjust=True, progress=False
                )
                df = pd.concat([df.iloc[:-14], df_new])
                df.to_csv(csv_f)
        else:
            df = yf.download(
                t, start=start, end=end,
                auto_adjust=True, progress=False
            )
            df.to_csv(csv_f)

        # ---------------- flatten columns safely -------------
        if isinstance(df.columns, pd.MultiIndex):
            # Which level contains the OHLCV words?
            lvl0 = df.columns.get_level_values(0)
            lvl1 = df.columns.get_level_values(1)
            if "Close" in lvl0:
                df.columns = lvl0           # keep level‑0
            elif "Close" in lvl1:
                df.columns = lvl1           # keep level‑1
            else:
                raise ValueError("Cannot locate 'Close' after yfinance download.")

        # ---------------- pick & rename -----------------------
        try:
            px  = df["Close"].rename(t)
        except KeyError as e:       # yfinance < 0.2 kept 'Adj Close'

        vol = df["Volume"].rename(f"{t}_VOL")

        # guarantee numeric
        px  = pd.to_numeric(px,  errors="coerce")

## 1\. 새 함수: `portfolio_table`

    python

    ###############################################################################

    7. 상세 포트폴리오 테이블 (달러 단위)

    ############################################################################### def portfolio_table(price_df, weight_df, start_capital=1_000_000, fee_pct=FEE_PCT): """ 반환값: DataFrame index = 거래일 컬럼 = [각 ETF 달러잔고] + [turnover_pct, fee_usd, nav_pre, nav_post]

    가정
▸ weight_df: TARGET 비중 (EOD 기준) — 이미 baseline_weights()로 생성
▸ 실제 매매는 다음 거래일 시가/종가(편의상 종가)에서 체결
"""
# 0) 준비
tickers = [c for c in weight_df.columns if not c.endswith("_VOL")]
w_tgt   = weight_df.shift().fillna(0)                     # 실행용 비중
rets    = price_df[tickers].pct_change().loc[w_tgt.index]

# 1) 매일 NAV·달러잔고·수수료 계산
nav_pre   = pd.Series(start_capital, index=w_tgt.index)   # 전일 종료 NAV
nav_post  = nav_pre.copy()
fee_usd   = pd.Series(0.0, index=w_tgt.index)
turnover  = pd.Series(0.0, index=w_tgt.index)

# DataFrame에 담을 달러 단위 포지션
pos_dollar = pd.DataFrame(0.0, index=w_tgt.index, columns=tickers)

for t in range(1, len(w_tgt)):
    # ===== t-1 종가 이후 =====
    prev_nav = nav_post.iloc[t-1]

    # (a) 전일 끝난 뒤 리밸런스 → 달러 거래·수수료
    dw        = (w_tgt.iloc[t] - w_tgt.iloc[t-1]).abs()
    turn_pct  = dw.sum()
    fee       = prev_nav * turn_pct * fee_pct

    turnover.iloc[t] = turn_pct
    fee_usd.iloc[t]  = fee

    # (b) 새로운 달러 포지션
    dol_pos = w_tgt.iloc[t] * prev_nav                       # 달러 단위
    pos_dollar.iloc[t] = dol_pos

    # (c) 하루동안 가격 변동
    day_ret = (dol_pos / prev_nav * rets.iloc[t]).sum()

    nav_pre.iloc[t]  = prev_nav
    nav_post.iloc[t] = prev_nav * (1 + day_ret) - fee

# 2) 결과 테이블 합치기
detail = pd.concat([pos_dollar.add_suffix("_USD"),
                    (w_tgt*100).add_suffix("_pct"),
                    turnover.rename("turnover_pct"),
                    fee_usd.rename("fee_usd"),
                    nav_pre.rename("nav_pre"),
                    nav_post.rename("nav_post")],
                   axis=1)

return detail

**배당·분할 반영**| `auto_adjust=True` 이면 종가가 이미 배당·액면분할을 보정해 주므로 **장기 수익률 왜곡** 이 없다.| *매매 체결가는 “실제 미조정 종가”*인데, 우리는 조정후 가격으로 사고판다고 가정하게 됨. 배당락(Ex‑dividend)일에는 **1일짜리 슬리피지** 가 생길 수 있다.| 백테스트는 그대로 두고, 실거래 시에는 **전일 종가 대비 다음날 시가(open)** 로 리밸런스하거나, 배당락일 ±1일 구간을 제외하고 영향도를 측정해 보는 것이 안전.  
**데이터 가용성**|  시장 종료 후 바로 업데이트되므로 일별 스트래티지 개발이 간편.| 장중 트레이딩(당일 종가 매매) 전략을 만들 경우 **미래 시점 데이터** 를 사용하게 되는 _룩어헤드 바이어스_ 가 발생함.| 스윙·포지셔닝(다음날 시가 체결) 목적이라면 문제 없음. 당일 종가 체결 모델이라면 **intraday 데이터**(예: 15 m OHLC) 로 검증 필요.  
**호가·체결가 격차**|  대형 ETF는 종가‑스프레드가 작아 보정 가격 ≒ 실제 종가.| 변동성 급등 구간, 저유동 ETF에는 **Bid‑Ask 스프레드** 가 0.1 % 이상 벌어질 때도 있음.| 비용(0.03 %) 모델에 **고정 슬리피지(예: 추가 0.02 %)**를 얹어 보수적으로 시뮬레이션.  
  
**결론**  
현재 같은 _다음 거래일 종가_ 체결 가정(backtest 시 weight를 1칸 `shift()`해 놓음)이라면 **Adjusted Close 사용 자체가 큰 문제는 없음**.  
다만,

  * **당일 종가 체결** 전략을 시도할 때는 룩어헤드 주의

  * **배당락일·고변동 구간** 영향도를 파악 후, 추가 슬리피지 또는 시가(open) 체결 모델로 전환

정도만 체크하면 됩니다.

### 2\. Gymnasium 환경 구현

    python

    import gymnasium as gym from gymnasium import spaces import numpy as np import pandas as pd

    class PortfolioEnv(gym.Env): """ observation : (n_features,) － 각 ETF의 기술지표 + 포트 현재 비중 action : (n_assets,) － 미가공 weight 벡터 (softmax로 0~1 정규화) reward : 에피소드 종료 시 최종 NAV(log수익) － 그 외 step은 0 """ metadata = {"render_modes": ["human"]}

    def __init__(self,
             price_df: pd.DataFrame,
             feat_df: pd.DataFrame,
             fee_pct: float = 3e-4,
             start_capital: float = 1_000_000,
             window: int = 1):
    super().__init__()
    self.price = price_df
    self.feat  = feat_df
    self.tickers = [c for c in price_df.columns if c in ETF_LIST]

    self.fee_pct = fee_pct
    self.start_capital = start_capital
    self.window = window

    # ───── Gym spaces ─────────────────────────────────
    n_assets   = len(self.tickers)
    n_features = feat_df.shape[1]

    self.action_space = spaces.Box(low=-np.inf, high=np.inf, shape=(n_assets,))
    self.observation_space = spaces.Box(
        low=-np.inf, high=np.inf,
        shape=(n_features + n_assets,)
    )
    # ───── 내부 상태 변수 ───────────────────────────────
    self.reset()

# ───────── 필수 메서드 ────────────────────────────────
def reset(self, *, seed=None, options=None):
    super().reset(seed=seed)
    self.idx = 0
    self.nav = self.start_capital
    self.weights = np.zeros(len(self.tickers))              # 초기 현금 100 %
    self._terminated = False
    return self._get_obs(), {}

def step(self, action):
    if self._terminated:
        raise RuntimeError("Episode already done")

    # 0. softmax → 0~1 가중치 & 합 = 1
    w_target = np.exp(action)
    w_target /= w_target.sum()

    # 1. 매매비용
    turnover = np.abs(w_target - self.weights).sum()
    fee = self.nav * turnover * self.fee_pct

    # 2. 하루 수익률
    if self.idx == 0:
        ret = 0.0
    else:
        day_ret = (self.price.iloc[self.idx] / self.price.iloc[self.idx - 1] - 1).values
        # 포트 수익률 = 가중합
        ret = np.dot(self.weights, day_ret)

    # 3. NAV 업데이트
    self.nav = self.nav * (1 + ret) - fee
    self.weights = w_target

    # 4. 인덱스 이동
    self.idx += 1
    done = self.idx >= len(self.price) - 1
    self._terminated = done

    # 5. 보상            ──────  (희소 보상) ──────
    reward = np.log(self.nav / self.start_capital) if done else 0.0
    info = {"nav": self.nav, "turnover": turnover, "fee": fee}

    return self._get_obs(), reward, done, False, info

# ───────── 헬퍼 ────────────────────────────────────
def _get_obs(self):
    feat_vals = self.feat.iloc[self.idx].values.astype(np.float32)
    obs = np.concatenate([feat_vals, self.weights]).astype(np.float32)
    return obs

def render(self):
    print(f"Step={self.idx} NAV={self.nav:,.2f} Weights={self.weights}")

    #### 설계 포인트

요소| 선택 이유  
---|---  
**행동(Action) 공간**|  선형 `[−∞, ∞]` 벡터 → softmax 변환으로 **‘합 = 1’·음수 불가** 제약 자동 만족.  
**보상**|  중간 단계에 0, 마지막 단계에 `log(NAV / 초기자본)` → 요청하신 “최종 수익만 리워드” 충족.  
**상태**|  기술지표 + 현 보유비중 → 에이전트가 **현재 포트 상태** 를 인지 가능.  
**수수료·턴오버**|  기존 `fee_pct=0.03 %` 적용.  
  
> ⚠️ _Sparse Reward_ : PPO·SAC 같은 정책 그래디언트 계열도 학습은 가능하지만, **에피소드 길이가 길수록** 탐색이 어려워집니다.
> 
>   * _Tip_  : 최종 보상 외에 “매일 log return ‑ λ × 수수료”를 추가하여 _dense shaping_ 을 쓰면 수렴 속도가 크게 빨라집니다.
