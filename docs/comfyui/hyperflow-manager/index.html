    좋아요! 아래는 PySide2/PySide6 겸용 으로 돌아가는 “HyperFlow(=ComfyUI) 매니저” 예시입니다.
    요구하신 동작(포트 찾기 → venv로 ComfyUI 실행 → Chrome 접속), UI 버튼(재시작/Chrome/종료), 트레이(닫기/최소화 시 트레이로, 경고 아이콘 메시지, 우클릭 메뉴 동일), 로그 뷰어까지 한 파일에 담았습니다.

    🔧 셸을 source venv/bin/activate로 활성화하는 대신, venv의 파이썬(예:venv/bin/python)을 직접 호출하는 방식이 훨씬 견고합니다(같은 효과). 아래 코드는 이 방법을 사용합니다.
    🖼 창/트레이 아이콘은 --icon 인자로 주거나, 코드 상단 ICON_PATH에 경로를 적어 주세요.

    사용 방법 (요약)

    python hyperflow_manager.py \
  --venv-python /path/to/venv/bin/python \
  --comfy-dir   /path/to/ComfyUI \
  --chrome      /path/to/google-chrome \
  --icon        /path/to/icon.png \
  --port        8188

    • --port를 생략하면 자동으로 사용 가능한 포트를 찾습니다.

    • ComfyUI는 python main.py --port <PORT> 로 실행한다고 가정합니다(필요시 인자 수정 가능).

    • Chrome은 http://127.0.0.1:<PORT>/ 로 뜹니다.

    코드: hyperflow_manager.py

    #!/usr/bin/env python3
# -*- coding: utf-8 -*-

import os
import sys
import json
import socket
import argparse
from pathlib import Path

# ----- PySide6 우선, 실패 시 PySide2 -----
try:
    from PySide6.QtCore import Qt, QEvent, QTimer, QProcess, QSize
    from PySide6.QtWidgets import (
        QApplication, QWidget, QPushButton, QVBoxLayout, QHBoxLayout, QLabel,
        QPlainTextEdit, QMessageBox, QFileDialog, QSystemTrayIcon, QMenu
    )
    from PySide6.QtGui import QIcon, QAction
    PYSIDE_VER = 6
except Exception:
    from PySide2.QtCore import Qt, QEvent, QTimer, QProcess, QSize
    from PySide2.QtWidgets import (
        QApplication, QWidget, QPushButton, QVBoxLayout, QHBoxLayout, QLabel,
        QPlainTextEdit, QMessageBox, QFileDialog, QSystemTrayIcon, QMenu
    )
    from PySide2.QtGui import QIcon, QAction
    PYSIDE_VER = 2

APP_NAME = "HyperFlow Manager"
CONFIG_PATH = Path.home() / ".hyperflow_manager.json"

# 기본값 (CLI/환경/설정파일로 덮어쓰기 가능)
DEFAULT_PREFERRED_PORT = 8188
ICON_PATH = ""  # 기본 아이콘 경로 없으면 시스템 기본
COMFY_DIR = ""  # ComfyUI 디렉토리 (main.py가 있는 곳)
VENV_PY = ""    # venv/bin/python 경로
CHROME_PATH = ""  # 크롬 실행 파일 경로 (사용자가 제공)

def load_config():
    if CONFIG_PATH.exists():
        try:
            return json.loads(CONFIG_PATH.read_text(encoding="utf-8"))
        except Exception:
            pass
    return {}

def save_config(d):
    try:
        CONFIG_PATH.write_text(json.dumps(d, indent=2, ensure_ascii=False), encoding="utf-8")
    except Exception:
        pass

def is_port_free(port: int, host="127.0.0.1") -> bool:
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.settimeout(0.2)
        return s.connect_ex((host, port)) != 0

def find_free_port(start_port: int = DEFAULT_PREFERRED_PORT, host="127.0.0.1") -> int:
    if start_port <= 0:
        # 완전 자동: OS 한테 포트 받기
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.bind((host, 0))
        port = s.getsockname()[1]
        s.close()
        return port
    # start_port부터 위로 스캔
    port = max(1, start_port)
    for _ in range(1000):
        if is_port_free(port, host=host):
            return port
        port += 1
    raise RuntimeError("사용 가능한 포트를 찾지 못했습니다.")

class HyperFlowManager(QWidget):
    def __init__(self, venv_py, comfy_dir, chrome_path, icon_path, preferred_port):
        super().__init__()
        self.venv_py = venv_py
        self.comfy_dir = comfy_dir
        self.chrome_path = chrome_path
        self.icon_path = icon_path
        self.preferred_port = preferred_port or DEFAULT_PREFERRED_PORT

        self.port = None
        self.comfy_proc = None  # QProcess
        self.chrome_proc = None  # QProcess
        self.tray = None
        self.tray_warned_once = False

        self.init_ui()
        self.init_tray()
        QApplication.setQuitOnLastWindowClosed(False)

        # 시작 시: 포트 찾고 → ComfyUI 실행 → Chrome 실행
        QTimer.singleShot(300, self.bootstrap_startup)

    # ---------------- UI ----------------
    def init_ui(self):
        self.setWindowTitle(APP_NAME)

        if self.icon_path and Path(self.icon_path).exists():
            self.setWindowIcon(QIcon(self.icon_path))

        self.port_label = QLabel("포트: -")
        self.status_label = QLabel("상태: 초기화 대기")
        self.log_view = QPlainTextEdit()
        self.log_view.setReadOnly(True)
        self.log_view.setMaximumBlockCount(10000)  # 로그 무한 증가 방지

        btn_restart = QPushButton("HyperFlow 재시작")
        btn_chrome  = QPushButton("Chrome 열기")
        btn_exit    = QPushButton("종료")

        btn_restart.clicked.connect(self.on_restart_clicked)
        btn_chrome.clicked.connect(self.on_chrome_clicked)
        btn_exit.clicked.connect(self.on_exit_clicked)

        # 파일 경로 확인용 작은 버튼들(선택사항)
        btn_pick_icon = QPushButton("아이콘 변경…")
        btn_pick_icon.clicked.connect(self.pick_icon)

        btn_pick_chrome = QPushButton("Chrome 경로 변경…")
        btn_pick_chrome.clicked.connect(self.pick_chrome)

        top_row = QHBoxLayout()
        top_row.addWidget(self.port_label)
        top_row.addStretch(1)
        top_row.addWidget(self.status_label)

        btn_row = QHBoxLayout()
        btn_row.addWidget(btn_restart)
        btn_row.addWidget(btn_chrome)
        btn_row.addWidget(btn_exit)

        path_row = QHBoxLayout()
        path_row.addWidget(btn_pick_icon)
        path_row.addWidget(btn_pick_chrome)
        path_row.addStretch(1)

        root = QVBoxLayout()
        root.addLayout(top_row)
        root.addLayout(btn_row)
        root.addLayout(path_row)
        root.addWidget(self.log_view)

        self.setLayout(root)
        self.resize(900, 600)

    def init_tray(self):
        self.tray = QSystemTrayIcon(self)
        if self.icon_path and Path(self.icon_path).exists():
            self.tray.setIcon(QIcon(self.icon_path))
        else:
            self.tray.setIcon(self.style().standardIcon(self.style().SP_ComputerIcon))
        self.tray.setToolTip(APP_NAME)

        menu = QMenu()
        act_show = QAction("창 열기", self)
        act_restart = QAction("HyperFlow 재시작", self)
        act_chrome  = QAction("Chrome 열기", self)
        act_exit    = QAction("종료", self)

        act_show.triggered.connect(self.show_main_window)
        act_restart.triggered.connect(self.on_restart_clicked)
        act_chrome.triggered.connect(self.on_chrome_clicked)
        act_exit.triggered.connect(self.on_exit_clicked)

        menu.addAction(act_show)
        menu.addSeparator()
        menu.addAction(act_restart)
        menu.addAction(act_chrome)
        menu.addSeparator()
        menu.addAction(act_exit)
        self.tray.setContextMenu(menu)

        self.tray.show()

    # ------------- 창/트레이 동작 -------------
    def changeEvent(self, event):
        # 최소화 시 트레이로
        if event.type() == QEvent.WindowStateChange:
            if self.isMinimized():
                QTimer.singleShot(0, self.hide)
                self.tray_warn("앱이 트레이로 이동했습니다.")
        super().changeEvent(event)

    def closeEvent(self, event):
        # X버튼 눌러도 종료 대신 트레이로
        event.ignore()
        self.hide()
        self.tray_warn("앱이 트레이로 이동했습니다.")

    def show_main_window(self):
        self.show()
        self.setWindowState(self.windowState() & ~Qt.WindowMinimized | Qt.WindowActive)
        self.activateWindow()

    def tray_warn(self, msg: str):
        # 경고 아이콘으로 트레이 메시지
        icon = QSystemTrayIcon.MessageIcon.Warning
        # 너무 자주 울리지 않게 기본 1회는 꼭 표시, 이후는 살짝 간격 제한 가능
        if not self.tray_warned_once:
            self.tray_warned_once = True
            self.tray.showMessage(APP_NAME, msg, icon, 3500)
        else:
            self.tray.showMessage(APP_NAME, msg, icon, 2500)

    # ------------- 초기 부팅 루틴 -------------
    def bootstrap_startup(self):
        try:
            if not self.port:
                self.port = find_free_port(self.preferred_port)
            self.port_label.setText(f"포트: {self.port}")
            self.append_log(f"[INFO] 선택된 포트: {self.port}")

            self.start_comfyui()
            # ComfyUI가 뜨는데 약간의 시간이 필요할 수 있음 → 크롬은 약간 지연 후 실행
            QTimer.singleShot(1200, self.on_chrome_clicked)
        except Exception as e:
            self.append_log(f"[ERROR] 초기화 실패: {e}")
            QMessageBox.critical(self, APP_NAME, f"초기화 실패: {e}")

    # ------------- 프로세스 제어 -------------
    def start_comfyui(self):
        if not self.venv_py or not Path(self.venv_py).exists():
            QMessageBox.critical(self, APP_NAME, "venv 파이썬 경로가 올바르지 않습니다.")
            return
        if not self.comfy_dir or not Path(self.comfy_dir).exists():
            QMessageBox.critical(self, APP_NAME, "ComfyUI 디렉토리 경로가 올바르지 않습니다.")
            return

        self.stop_comfyui(force=True)

        self.comfy_proc = QProcess(self)
        self.comfy_proc.setProgram(self.venv_py)
        self.comfy_proc.setWorkingDirectory(self.comfy_dir)

        # 필요시 main.py 대신 특정 진입점으로 바꾸세요.
        args = ["main.py", "--port", str(self.port)]
        # 외부 접속이 필요하면 다음 줄 주석 해제:
        # args += ["--listen", "0.0.0.0"]
        self.comfy_proc.setArguments(args)

        # 표준 출력/에러 합치기
        self.comfy_proc.setProcessChannelMode(QProcess.MergedChannels)
        self.comfy_proc.readyReadStandardOutput.connect(self.on_comfy_output)
        self.comfy_proc.finished.connect(self.on_comfy_finished)

        self.append_log(f"[INFO] HyperFlow(ComfyUI) 실행: {self.venv_py} {' '.join(args)}")
        self.comfy_proc.start()

        if not self.comfy_proc.waitForStarted(5000):
            self.append_log("[ERROR] HyperFlow 시작 실패 (시간 초과)")
            QMessageBox.critical(self, APP_NAME, "HyperFlow 시작 실패")
            return

        self.status_label.setText("상태: HyperFlow 실행 중")

    def stop_comfyui(self, force=False):
        if self.comfy_proc is not None:
            if self.comfy_proc.state() != QProcess.NotRunning:
                self.append_log("[INFO] HyperFlow 종료 중…")
                self.comfy_proc.terminate()
                if not self.comfy_proc.waitForFinished(2000) and force:
                    self.comfy_proc.kill()
                    self.comfy_proc.waitForFinished(1000)
            self.comfy_proc = None
            self.status_label.setText("상태: HyperFlow 정지")

    def on_comfy_output(self):
        if not self.comfy_proc:
            return
        data = bytes(self.comfy_proc.readAllStandardOutput()).decode(errors="ignore")
        if data:
            self.append_log(data.rstrip("\n"))

    def on_comfy_finished(self, code, status):
        self.append_log(f"[INFO] HyperFlow 종료(code={code}, status={status})")
        self.status_label.setText("상태: HyperFlow 정지")

    def start_chrome(self):
        if not self.chrome_path or not Path(self.chrome_path).exists():
            # 경로가 없으면 시스템 기본 브라우저로라도 열기
            import webbrowser
            url = f"http://127.0.0.1:{self.port}/"
            self.append_log(f"[WARN] Chrome 경로 없음 → 기본 브라우저로 열기: {url}")
            webbrowser.open(url)
            return

        self.stop_chrome()

        url = f"http://127.0.0.1:{self.port}/"
        self.chrome_proc = QProcess(self)
        self.chrome_proc.setProgram(self.chrome_path)
        # 새 프로필을 쓰면 LSF 서버 환경에서도 충돌 적음(선택)
        user_dir = f"/tmp/hyperflow_profile_{self.port}"
        args = ["--new-window", url, f"--user-data-dir={user_dir}"]
        self.chrome_proc.setArguments(args)
        self.chrome_proc.setProcessChannelMode(QProcess.MergedChannels)
        self.chrome_proc.readyReadStandardOutput.connect(self.on_chrome_output)
        self.chrome_proc.finished.connect(self.on_chrome_finished)

        self.append_log(f"[INFO] Chrome 실행: {self.chrome_path} {' '.join(args)}")
        self.chrome_proc.start()
        if not self.chrome_proc.waitForStarted(5000):
            self.append_log("[ERROR] Chrome 시작 실패 (시간 초과)")
            QMessageBox.warning(self, APP_NAME, "Chrome 시작 실패")

    def stop_chrome(self):
        if self.chrome_proc is not None:
            if self.chrome_proc.state() != QProcess.NotRunning:
                self.append_log("[INFO] Chrome 종료 중…")
                self.chrome_proc.terminate()
                if not self.chrome_proc.waitForFinished(1500):
                    self.chrome_proc.kill()
                    self.chrome_proc.waitForFinished(800)
            self.chrome_proc = None

    def on_chrome_output(self):
        if not self.chrome_proc:
            return
        data = bytes(self.chrome_proc.readAllStandardOutput()).decode(errors="ignore")
        if data:
            self.append_log("[Chrome] " + data.rstrip("\n"))

    def on_chrome_finished(self, code, status):
        self.append_log(f"[INFO] Chrome 종료(code={code}, status={status})")

    # ------------- 버튼 동작 -------------
    def on_restart_clicked(self):
        self.append_log("[ACTION] HyperFlow 재시작")
        # 같은 포트로 재시작 시도. 실패하면 새 포트 확보
        self.stop_comfyui(force=True)
        try:
            if not is_port_free(self.port):
                self.port = find_free_port(self.port + 1)
                self.port_label.setText(f"포트: {self.port}")
        except Exception as e:
            self.append_log(f"[WARN] 포트 재확보 실패: {e}")
        self.start_comfyui()

    def on_chrome_clicked(self):
        self.append_log("[ACTION] Chrome 열기")
        self.start_chrome()

    def on_exit_clicked(self):
        self.append_log("[ACTION] 종료")
        self.stop_chrome()
        self.stop_comfyui(force=True)
        QApplication.quit()

    # ------------- 보조 -------------
    def append_log(self, text: str):
        self.log_view.appendPlainText(text)
        # 자동 스크롤
        cursor = self.log_view.textCursor()
        cursor.movePosition(cursor.End)
        self.log_view.setTextCursor(cursor)

    def pick_icon(self):
        path, _ = QFileDialog.getOpenFileName(self, "아이콘 선택", str(Path.home()), "이미지 파일 (*.png *.ico *.jpg *.bmp)")
        if path:
            self.icon_path = path
            self.setWindowIcon(QIcon(self.icon_path))
            if self.tray:
                self.tray.setIcon(QIcon(self.icon_path))
            self.append_log(f"[INFO] 아이콘 변경: {path}")
            self.persist_settings()

    def pick_chrome(self):
        path, _ = QFileDialog.getOpenFileName(self, "Chrome 실행 파일 선택", "/usr/bin", "실행 파일 (*)")
        if path:
            self.chrome_path = path
            self.append_log(f"[INFO] Chrome 경로 변경: {path}")
            self.persist_settings()

    def persist_settings(self):
        cfg = load_config()
        cfg.update({
            "venv_py": self.venv_py,
            "comfy_dir": self.comfy_dir,
            "chrome_path": self.chrome_path,
            "icon_path": self.icon_path,
            "preferred_port": self.preferred_port,
        })
        save_config(cfg)

def parse_args():
    p = argparse.ArgumentParser(description="HyperFlow(ComfyUI) Manager")
    p.add_argument("--venv-python", dest="venv_py", type=str, default="")
    p.add_argument("--comfy-dir", dest="comfy_dir", type=str, default="")
    p.add_argument("--chrome", dest="chrome_path", type=str, default="")
    p.add_argument("--icon", dest="icon_path", type=str, default="")
    p.add_argument("--port", dest="port", type=int, default=DEFAULT_PREFERRED_PORT)
    return p.parse_args()

def main():
    # 설정 적재 우선순위: CLI > 저장파일 > 환경변수 > 기본값
    args = parse_args()
    stored = load_config()

    venv_py = args.venv_py or stored.get("venv_py") or os.environ.get("HFM_VENV_PY", VENV_PY)
    comfy_dir = args.comfy_dir or stored.get("comfy_dir") or os.environ.get("HFM_COMFY_DIR", COMFY_DIR)
    chrome_path = args.chrome_path or stored.get("chrome_path") or os.environ.get("HFM_CHROME_PATH", CHROME_PATH)
    icon_path = args.icon_path or stored.get("icon_path") or os.environ.get("HFM_ICON_PATH", ICON_PATH)
    port = args.port or stored.get("preferred_port") or DEFAULT_PREFERRED_PORT

    app = QApplication(sys.argv)
    app.setApplicationName(APP_NAME)

    w = HyperFlowManager(venv_py=venv_py, comfy_dir=comfy_dir,
                         chrome_path=chrome_path, icon_path=icon_path,
                         preferred_port=port)
    w.show()
    sys.exit(app.exec())

if __name__ == "__main__":
    main()

    구현 체크리스트

    • 시작 시:

      1. 사용 가능한 포트 자동 탐색 →

      2. venv/bin/python main.py --port <포트> 로 HyperFlow(ComfyUI) 실행 →

      3. 잠시 대기 후 Chrome(지정 경로)로 http://127.0.0.1:<포트>/ 접속.

    • UI:

      • HyperFlow 재시작 , Chrome 열기 , 종료 버튼.

      • 로그: QProcess의 stdout/stderr를 QPlainTextEdit 에 실시간 표시.

      • 아이콘: --icon 혹은 UI에서 “아이콘 변경…”으로 교체 → 창과 트레이 모두 반영.

    • 트레이:

      • X/최소화 시 트레이로 숨김 + 경고 아이콘 으로 풍선 메시지.

      • 우클릭 메뉴: 창 열기 / HyperFlow 재시작 / Chrome 열기 / 종료.

    csh를 PySide 위젯 안에 “그대로” 넣을 수 있나?

    • 직접 내장된 csh 위젯은 없습니다. PySide/Qt 기본 위젯에 셸을 “임베드”하는 기능은 없어요.

    • 선택지:

      1. 로그만 필요 : 지금 예시처럼 QProcess로 프로세스 실행 후 stdout/stderr를 텍스트 위젯 에 스트리밍 표시(가장 간단/안정적).

      2. 대화형 터미널 임베드 : 리눅스라면 QTermWidget(KDE Konsole 컴포넌트, libqtermwidget) 같은 서드파티 위젯 을 연동할 수 있습니다(빌드/패키지 필요). 이 경우 csh도 터미널 내에서 사용할 수 있어요.

      3. 준-대화형 : 파이썬 pty/pexpect로 의사 터미널을 만들어 위젯과 연결(복잡하고 호환성 이슈 있음).

    • 결론: 로그 표시 가 목적이면 지금 방식이 가장 현실적이고, 완전한 터미널 을 원하면 QTermWidget 채택을 추천합니다.

    필요한 부분만 바꾸면 바로 사용 가능

    • --venv-python : .../venv/bin/python

    • --comfy-dir : .../ComfyUI (여기에 main.py)

    • --chrome : google-chrome(혹은 chromium, 경로 명시)

    • --icon : 원하는 이미지 파일(.png 권장)

    • --port : 고정 포트 쓰려면 지정, 아니면 생략해서 자동 탐색

    원하시면 HyperFlow 이름, 로그 포맷, ComfyUI 인자(--listen, --multi-user, 등)도 추가해 드릴게요.
