    ComfyUI custom node 추가(glob)

    2025-09-13

    ComfyUI에 새로운 Custom node를 추가하고 싶어. python의 glob와 비슷한 역할을 하는 노드를 만들고 싶어. custom_nodes/hyperflow-core 내부에 추가할 거야. - Node name: HFFindPathnames - Category: HyperFlow - Node Display Name: HF - Find Pathnames(glob) Inputs: - pathname(str) - 검색 패턴 - root_dir(str, Optional) - 주어지지 않는다면 현재 디렉토리를 사용 - recursive(bool, default: False) - “” 패턴이 모든 디렉터리/파일과 매칭됨(하위 포함) - include_hidden(bool, default: False) - “” 패턴이 숨겨진 파일도 매칭됨 outputs: - match된 path name list

    좋아! 요청하신 사양으로 ComfyUI 커스텀 노드를 깔끔하게 만들어드릴게요.
    (아래 코드는 그대로 저장하면 바로 동작합니다.)

    1) 설계 요약 (step-by-step)

    1. 입력 위젯

      • pathname: STRING (필수) – **/*.png 같은 glob 패턴

      • root_dir: STRING (옵션) – 비우면 현재 작업 디렉터리 사용

      • recursive: BOOLEAN (기본 False)

      • include_hidden: BOOLEAN (기본 False)

    2. 출력

      • STRING 리스트 한 개(매칭된 경로들).
        Comfy가 이 리스트를 “여러 아이템”으로 다루도록 OUTPUT_IS_LIST = (True,)를 사용합니다. docs.comfy.org

    3. 동작

      • glob.iglob()을 기본 사용.

      • Python 3.11+ 환경이면 include_hidden 파라미터를 그대로 활용.

      • 3.11 미만일 때도 **가 들어간 패턴에 대해 숨김 포함을 최대한 보완(패턴 변형)해서 수집.

      • root_dir가 주어지면 상대 패턴은 해당 루트 기준으로 탐색.

      • 경로/환경변수(~, %VAR%) 확장 지원.

    4. 타입 정의

      • 입력 타입: STRING, BOOLEAN (공식 문서의 기본 타입 표준을 따름). docs.comfy.org

    2) 설치 위치

    다음 파일로 저장하세요:

    custom_nodes/hyperflow-core/hf_find_pathnames.py

    3) 코드

    # custom_nodes/hyperflow-core/hf_find_pathnames.py

import os
import glob
import inspect
from itertools import product

class HFFindPathnames:
    """
    HF - Find Pathnames(glob)
    - Python glob과 유사한 패턴 매칭
    - include_hidden(True)일 때: Python 3.11+이면 glob의 include_hidden을 사용.
      3.11 미만이면 '**' 패턴에 한해 숨김 포함 변형 패턴을 추가로 탐색(최대한의 호환).
    """

    @classmethod
    def INPUT_TYPES(s):
        return {
            "required": {
                "pathname": ("STRING", {
                    "default": "",
                    "placeholder": "예: **/*.png"
                }),
            },
            "optional": {
                "root_dir": ("STRING", {
                    "default": "",
                    "placeholder": "비우면 현재 디렉토리"
                }),
                "recursive": ("BOOLEAN", {
                    "default": False,
                    "label_on": "recursive",
                    "label_off": "not recursive"
                }),
                "include_hidden": ("BOOLEAN", {
                    "default": False,
                    "label_on": "include .dot files",
                    "label_off": "exclude .dot files"
                }),
            }
        }

    RETURN_TYPES = ("STRING",)
    RETURN_NAMES = ("pathnames",)
    # 리스트 출력임을 명시 (다음 노드가 리스트를 순차 처리하도록)
    OUTPUT_IS_LIST = (True,)
    FUNCTION = "find"
    CATEGORY = "HyperFlow"

    def _expand(self, s: str) -> str:
        # 환경변수/홈(~) 확장 + 정규화
        return os.path.normpath(os.path.expandvars(os.path.expanduser(s)))

    def _effective_pattern(self, pathname: str, root_dir: str | None) -> str:
        pathname = self._expand(pathname)
        if os.path.isabs(pathname):
            return pathname
        base = self._expand(root_dir) if root_dir else os.getcwd()
        return os.path.join(base, pathname)

    def _iglob(self, pattern: str, recursive: bool, include_hidden: bool):
        """Python 3.11+이면 include_hidden 사용, 아니면 호환 경로."""
        sig = inspect.signature(glob.iglob)
        params = sig.parameters
        if "include_hidden" in params:  # Python 3.11+
            return glob.iglob(pattern, recursive=bool(recursive), include_hidden=bool(include_hidden))
        else:
            # 3.11 미만: 기본 동작(숨김 제외)만 제공
            return glob.iglob(pattern, recursive=bool(recursive))

    def _hidden_variants_for_legacy(self, pattern: str, limit: int = 3):
        """
        3.11 미만 환경에서 include_hidden=True + '**' 패턴일 때,
        '**'를 '**/.*'로 치환한 변형을 추가 생성해 숨김 파일/폴더를 최대한 포괄.
        (첫 limit개 '**'에 대해서만 조합 생성하여 폭발 방지)
        """
        if "**" not in pattern:
            return []

        parts = pattern.split("**")
        n = len(parts) - 1
        if n <= 0:
            return []

        n_use = min(n, limit)
        prefix_parts = parts[:n_use + 1]
        suffix = "**".join(parts[n_use:]) if n_use < n else parts[-1]

        variants = []
        for combo in product([False, True], repeat=n_use):
            s = prefix_parts[0]
            for i, use_hidden in enumerate(combo, 1):
                s += ("**/.*" if use_hidden else "**") + prefix_parts[i]
            variants.append(s + suffix)
        return variants

    def find(self, pathname: str, root_dir: str = "", recursive: bool = False, include_hidden: bool = False):
        if not isinstance(pathname, str) or pathname.strip() == "":
            return ([],)

        root = root_dir.strip() or None
        pattern = self._effective_pattern(pathname, root)

        matches = set()

        # 1) 기본 패턴
        for p in self._iglob(pattern, recursive, include_hidden):
            matches.add(os.path.normpath(p))

        # 2) (레거시 호환) include_hidden=True + '**'일 때 변형 패턴도 검색
        if include_hidden and recursive and ("**" in pathname):
            for v in self._hidden_variants_for_legacy(pattern):
                for p in self._iglob(v, recursive, include_hidden=False):
                    matches.add(os.path.normpath(p))

        # 정렬(Windows는 대소문자 무시 정렬)
        if os.name == "nt":
            result = sorted(matches, key=lambda x: x.lower())
        else:
            result = sorted(matches)

        return (result,)

NODE_CLASS_MAPPINGS = {
    "HFFindPathnames": HFFindPathnames,
}

NODE_DISPLAY_NAME_MAPPINGS = {
    "HFFindPathnames": "HF - Find Pathnames(glob)",
}

    4) 사용 방법

    1. ComfyUI 재시작 → HyperFlow 카테고리에서 “HF - Find Pathnames(glob)” 검색.

    2. 예시

      • 현재 작업 폴더에서 PNG 모두:
        pathname="**/*.png", recursive=True

      • 특정 루트 기준으로 .txt 전부(숨김 포함):
        pathname="**/*.txt", root_dir="/data/project", recursive=True, include_hidden=True

    3. 출력은 문자열 리스트 이므로, 그 다음 노드가 리스트를 “순차 처리”할 수 있습니다(OUTPUT_IS_LIST 설정 덕분). docs.comfy.org

    참고(공식 문서)

    • 리스트 출력 처리(OUTPUT_IS_LIST) – ComfyUI 개발 문서 “Data lists” 섹션. docs.comfy.org

    • 입력 타입(STRING/BOOLEAN 등) 정의 – “Datatypes” 섹션. docs.comfy.org

    필요하면 RETURN_NAMES 변경(예: "matches")이나, UI placeholder/라벨 문구를 더 다듬어드릴게요.

    그런데 String List 말고 줄바꿈으로 Join된 하나의 String으로 나와도 편리할 것 같아. 이런 출력을 하나 추가해줘.

    좋은 아이디어예요! “줄바꿈으로 합친 하나의 문자열” 출력을 추가 만 해서, 기존 리스트 출력은 그대로 유지하겠습니다.

    변경 요약 (step-by-step)

    1. 출력 추가

    • 기존: ("STRING",) 1개 (리스트/다중항목)

    • 변경: ("STRING", "STRING") 2개

      • 1번 출력: pathnames (리스트, 기존과 동일)

      • 2번 출력: joined_text (단일 문자열, \n로 join)

    1. 리스트 표기
    • OUTPUT_IS_LIST = (True, False) 로 업데이트
    1. 리턴 값
    • return (result, "\n".join(result))

    교체 코드

    아래 내용을 기존 파일(custom_nodes/hyperflow-core/hf_find_pathnames.py)에 그대로 덮어쓰세요.

    # custom_nodes/hyperflow-core/hf_find_pathnames.py

import os
import glob
import inspect
from itertools import product

class HFFindPathnames:
    """
    HF - Find Pathnames(glob)
    - Python glob과 유사한 패턴 매칭
    - include_hidden(True)일 때: Python 3.11+이면 glob의 include_hidden을 사용.
      3.11 미만이면 '**' 패턴에 한해 숨김 포함 변형 패턴을 추가 탐색.
    - 출력:
      (1) pathnames: STRING 리스트 (OUTPUT_IS_LIST=True)
      (2) joined_text: 줄바꿈으로 합친 하나의 STRING
    """

    @classmethod
    def INPUT_TYPES(s):
        return {
            "required": {
                "pathname": ("STRING", {
                    "default": "",
                    "placeholder": "예: **/*.png"
                }),
            },
            "optional": {
                "root_dir": ("STRING", {
                    "default": "",
                    "placeholder": "비우면 현재 디렉토리"
                }),
                "recursive": ("BOOLEAN", {
                    "default": False,
                    "label_on": "recursive",
                    "label_off": "not recursive"
                }),
                "include_hidden": ("BOOLEAN", {
                    "default": False,
                    "label_on": "include .dot files",
                    "label_off": "exclude .dot files"
                }),
            }
        }

    # 출력 두 개: (리스트 STRING, 단일 STRING)
    RETURN_TYPES = ("STRING", "STRING")
    RETURN_NAMES = ("pathnames", "joined_text")
    OUTPUT_IS_LIST = (True, False)
    FUNCTION = "find"
    CATEGORY = "HyperFlow"

    def _expand(self, s: str) -> str:
        # 환경변수/홈(~) 확장 + 정규화
        return os.path.normpath(os.path.expandvars(os.path.expanduser(s)))

    def _effective_pattern(self, pathname: str, root_dir: str | None) -> str:
        pathname = self._expand(pathname)
        if os.path.isabs(pathname):
            return pathname
        base = self._expand(root_dir) if root_dir else os.getcwd()
        return os.path.join(base, pathname)

    def _iglob(self, pattern: str, recursive: bool, include_hidden: bool):
        """Python 3.11+이면 include_hidden 사용, 아니면 호환 경로."""
        sig = inspect.signature(glob.iglob)
        params = sig.parameters
        if "include_hidden" in params:  # Python 3.11+
            return glob.iglob(pattern, recursive=bool(recursive), include_hidden=bool(include_hidden))
        else:
            # 3.11 미만: 기본 동작(숨김 제외)만 제공
            return glob.iglob(pattern, recursive=bool(recursive))

    def _hidden_variants_for_legacy(self, pattern: str, limit: int = 3):
        """
        3.11 미만 환경에서 include_hidden=True + '**' 패턴일 때,
        '**'를 '**/.*'로 치환한 변형을 추가 생성해 숨김 파일/폴더를 최대한 포괄.
        (첫 limit개 '**'에 대해서만 조합 생성하여 폭발 방지)
        """
        if "**" not in pattern:
            return []

        parts = pattern.split("**")
        n = len(parts) - 1
        if n <= 0:
            return []

        n_use = min(n, limit)
        prefix_parts = parts[:n_use + 1]
        suffix = "**".join(parts[n_use:]) if n_use < n else parts[-1]

        variants = []
        for combo in product([False, True], repeat=n_use):
            s = prefix_parts[0]
            for i, use_hidden in enumerate(combo, 1):
                s += ("**/.*" if use_hidden else "**") + prefix_parts[i]
            variants.append(s + suffix)
        return variants

    def find(self, pathname: str, root_dir: str = "", recursive: bool = False, include_hidden: bool = False):
        if not isinstance(pathname, str) or pathname.strip() == "":
            # 빈 입력: 두 출력 모두 빈 형태
            return ([], "")

        root = root_dir.strip() or None
        pattern = self._effective_pattern(pathname, root)

        matches = set()

        # 1) 기본 패턴
        for p in self._iglob(pattern, recursive, include_hidden):
            matches.add(os.path.normpath(p))

        # 2) (레거시 호환) include_hidden=True + '**'일 때 변형 패턴도 검색
        if include_hidden and recursive and ("**" in pathname):
            for v in self._hidden_variants_for_legacy(pattern):
                for p in self._iglob(v, recursive, include_hidden=False):
                    matches.add(os.path.normpath(p))

        # 정렬(Windows는 대소문자 무시 정렬)
        if os.name == "nt":
            result = sorted(matches, key=lambda x: x.lower())
        else:
            result = sorted(matches)

        joined = "\n".join(result)
        return (result, joined)

    "HFFindPathnames": HFFindPathnames,
}

NODE_DISPLAY_NAME_MAPPINGS = {
    "HFFindPathnames": "HF - Find Pathnames(glob)",
}

    • 워크플로에서 리스트 처리 는 1번 출력(pathnames)을 쓰고,

      (예: Text Save, CLIPTextEncode 같은 문자열 단일 입력 노드)
