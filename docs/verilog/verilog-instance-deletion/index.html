    Verilog instance deletion

    2025-08-22

    Developer: - 목표: Verilog module netlist가 제공되었을 때, 파이썬을 사용하여 해당 module 내부에서 특정 이름을 가진 instance들을 삭제하는 기능을 구현합니다. - 작업 절차: 1. 입력 파일로 Verilog module netlist를 파싱합니다. 2. 지정된 이름(리스트 또는 패턴 가능)을 가진 instance들을 module 내에서 찾아 삭제합니다. 3. 삭제 후, 수정된 netlist를 파일 또는 문자열로 출력합니다. - 요구사항 및 한계: - instance 삭제 로직만 구현합니다(삽입/수정 없음). - netlist의 기본 파싱, instance 탐색, 문자열 또는 파일 입출력까지 지원합니다. - 복잡한 netlist 구조(첨단 macro, conditional include)는 지원하지 않습니다. - 출력: - 예시 코드 및 간단한 사용 방법(주석 포함) - 삭제 전후 netlist 예시(선택 사항) - 문서화: - 주요 함수와 삭제 로직에 설명 주석을 추가합니다.     Here’s the plan 👇

    • Clarify the scope: delete instances by exact names or regex patterns, inside selected modules or all modules.

    • Parse a Verilog netlist into module blocks (basic, comment-safe), keeping headers/footers intact.

    • Detect instantiation statements (supports optional #(...) params and multi-instance forms).

    • Remove matched instances (drop whole statement if all instances removed; otherwise rebuild it).

    • Output the modified netlist and a small report (what was deleted / left).

    Validation after key steps:

    • Parsing → confirm we found modules; if none, stop and report.

    • Deletion → confirm target instance names/patterns no longer appear textually in the affected modules (basic string/regex check).

    ✅ Minimal, well-commented Python utility

    import re
from typing import Iterable, List, Optional, Tuple, Dict, Pattern

VERILOG_KEYWORDS = {
    "module","endmodule","input","output","inout","wire","tri","supply0","supply1",
    "reg","logic","assign","always","initial","generate","endgenerate","begin","end",
    "parameter","localparam","typedef","function","endfunction","task","endtask"
}

def strip_comments(text: str) -> str:
    """
    Remove // line-comments and /* block-comments */.
    Note: string literals are ignored for simplicity (OK for basic netlists).
    """
    # Remove block comments
    text = re.sub(r"/\*.*?\*/", "", text, flags=re.S)
    # Remove line comments
    text = re.sub(r"//.*?$", "", text, flags=re.M)
    return text

def find_module_blocks(text: str) -> List[Tuple[str, int, int]]:
    """
    Return a list of (module_name, start_index, end_index) covering 'module ... endmodule'.
    """
    blocks = []
    for m in re.finditer(r"\bmodule\b.*?\bendmodule\b", text, flags=re.S):
        block = m.group(0)
        name_m = re.match(r"\s*module\s+([A-Za-z_]\w*)", block)
        name = name_m.group(1) if name_m else "<unknown>"
        blocks.append((name, m.start(), m.end()))
    return blocks

def split_module_parts(block_text: str) -> Tuple[str, str, str]:
    """
    Split a single module block into (header_including_semicolon, body, 'endmodule').
    We locate the first ';' after 'module' as the end of the header.
    """
    end_idx = block_text.find("endmodule")
    if end_idx == -1:
        return block_text, "", ""  # malformed, but fail gracefully
    header_end = block_text.find(";")
    if header_end == -1:
        # No header semicolon (unlikely). Treat all as header.
        return block_text[:end_idx], "", "endmodule"
    header = block_text[:header_end + 1]
    body = block_text[header_end + 1:end_idx]
    return header, body, "endmodule"

def skip_ws(s: str, i: int) -> int:
    while i < len(s) and s[i].isspace():
        i += 1
    return i

def parse_balanced(s: str, i: int, open_ch: str, close_ch: str) -> int:
    """
    Given s[i] == open_ch, consume until the matching close_ch and return the index AFTER it.
    Raise ValueError on mismatch.
    """
    if i >= len(s) or s[i] != open_ch:
        raise ValueError("Expected %r at position %d" % (open_ch, i))
    depth, i0 = 1, i + 1
    i = i0
    while i < len(s):
        c = s[i]
        if c == open_ch:
            depth += 1
        elif c == close_ch:
            depth -= 1
            if depth == 0:
                return i + 1
        i += 1
    raise ValueError("Unbalanced %r/%r starting at %d" % (open_ch, close_ch, i0 - 1))

def strip_leading_attribute(stmt: str) -> str:
    """
    Remove leading synthesis attributes like: (* keep = "true" *).
    """
    s = stmt.lstrip()
    while s.startswith("(*"):
        j = s.find("*)")
        if j == -1:
            break
        s = s[j + 2:].lstrip()
    return s

def first_token(s: str, i: int) -> Tuple[Optional[str], int]:
    m = re.match(r"[A-Za-z_]\w*", s[i:])
    if not m:
        return None, i
    tok = m.group(0)
    return tok, i + len(tok)

def looks_like_instantiation(stmt: str) -> bool:
    """
    Heuristic check: first token must be a non-keyword identifier, and the shape must
    allow optional '#(...)' then an identifier followed by '(' ... ')'.
    """
    s = strip_leading_attribute(stmt)
    i = skip_ws(s, 0)
    tok, i = first_token(s, i)
    if not tok or tok in VERILOG_KEYWORDS:
        return False
    # Optional parameterization: #( ... )
    i = skip_ws(s, i)
    if i < len(s) and s[i] == '#':
        i += 1
        i = skip_ws(s, i)
        try:
            i = parse_balanced(s, i, '(', ')')
        except ValueError:
            return False
    i = skip_ws(s, i)
    # Next must look like an instance name '('
    name, j = first_token(s, i)
    if not name:
        return False
    j = skip_ws(s, j)
    # Optional array range [..]
    if j < len(s) and s[j] == '[':
        try:
            j = parse_balanced(s, j, '[', ']')
        except ValueError:
            return False
        j = skip_ws(s, j)
    return (j < len(s) and s[j] == '(')

def parse_instance_statement(stmt: str) -> Optional[Dict]:
    """
    Parse a (possibly multi-instance) instantiation statement into its pieces.

    Returns:
        {
          "cell": str,
          "param_text": str ('' or like '#(...)'),
          "instances": List[{"name": str, "text": str}],
        }
    or None if parsing fails / not an instantiation.
    """
    if not looks_like_instantiation(stmt):
        return None

    original = stmt
    s = strip_leading_attribute(stmt)
    i = skip_ws(s, 0)

    cell, i = first_token(s, i)
    i = skip_ws(s, i)

    # Optional '#(...)'
    param_text = ""
    if i < len(s) and s[i] == '#':
        start = i
        i += 1
        i = skip_ws(s, i)
        end = parse_balanced(s, i, '(', ')')
        param_text = s[start:end]
        i = end
        i = skip_ws(s, i)

    # Collect instances until ';'
    instances = []
    # Find the trailing ';' to bound scanning
    try:
        end_stmt = max(idx for idx, ch in enumerate(s) if ch == ';') + 1
    except ValueError:
        end_stmt = len(s)

    while i < end_stmt:
        i = skip_ws(s, i)
        name, j = first_token(s, i)
        if not name:
            break
        i = j
        i = skip_ws(s, i)

        # Optional array range
        if i < len(s) and s[i] == '[':
            i = parse_balanced(s, i, '[', ']')
            i = skip_ws(s, i)

        # Port list
        if i >= len(s) or s[i] != '(':
            # Not a valid instance group -> give up
            return None
        inst_start = j - len(name)  # include the instance name start
        end_ports = parse_balanced(s, i, '(', ')')
        i = end_ports
        # Capture the raw text from name through the end of ')'
        inst_text = s[inst_start:end_ports]
        i = skip_ws(s, i)

        # If there's a comma, this statement has more instances
        if i < len(s) and s[i] == ',':
            # Include any following whitespace when we advance
            i += 1
        instances.append({"name": name, "text": inst_text})

        # Move on
        i = skip_ws(s, i)
        if i < len(s) and s[i] == ';':
            break

    if not instances:
        return None

    return {"cell": cell, "param_text": param_text, "instances": instances, "original": original}

def delete_instances_in_stmt(parsed: Dict,
                             names: Iterable[str],
                             patterns: List[Pattern]) -> Tuple[bool, str, List[str]]:
    """
    From a parsed instantiation, drop matching instances.
    Returns: (changed?, new_stmt_text, deleted_names)
    """
    to_delete = set(names) if names else set()
    deleted = []
    kept_texts = []
    for inst in parsed["instances"]:
        hit = inst["name"] in to_delete or any(p.search(inst["name"]) for p in patterns)
        if hit:
            deleted.append(inst["name"])
        else:
            kept_texts.append(inst["text"])

    if not deleted:
        return False, parsed["original"], []

    if not kept_texts:
        # Whole statement removed
        return True, "", deleted

    # Rebuild: <cell><param?> <instA(...)>, <instB(...>) ;
    rebuilt = f'{parsed["cell"]}{parsed["param_text"]} ' + ", ".join(kept_texts) + ";"
    return True, rebuilt, deleted

def remove_instances_from_verilog(
    verilog_text: str,
    instance_names: Optional[Iterable[str]] = None,
    instance_patterns: Optional[Iterable[str]] = None,
    limit_to_modules: Optional[Iterable[str]] = None,
) -> Tuple[str, Dict]:
    """
    Delete instances by name/pattern from selected modules (or all modules if None).

    Args:
        verilog_text: Entire netlist text.
        instance_names: Exact instance names to remove.
        instance_patterns: Regex patterns (python re) for names to remove.
        limit_to_modules: Only mutate these module names (others preserved verbatim).

    Returns: (modified_text, report_dict)
        report_dict = {
            "modules_parsed": [names...],
            "modules_edited": [names...],
            "deleted_instances": {module_name: [inst_names...]},
            "not_found": [requested_name that was not matched],
        }
    """
    names = list(instance_names or [])
    patterns = [re.compile(p) for p in (instance_patterns or [])]
    text = strip_comments(verilog_text)

    blocks = find_module_blocks(text)
    report = {
        "modules_parsed": [b[0] for b in blocks],
        "modules_edited": [],
        "deleted_instances": {},
        "not_found": [],
    }
    if not blocks:
        return verilog_text, report  # nothing to do

    out_parts = []
    cursor = 0
    global_deleted = set()

    for mod_name, s, e in blocks:
        # copy pre-module region unchanged
        out_parts.append(verilog_text[cursor:s])

        block = verilog_text[s:e]
        header, body, end_kw = split_module_parts(block)
        if limit_to_modules and mod_name not in set(limit_to_modules):
            # Leave this module as-is
            out_parts.append(block)
            cursor = e
            continue

        # Work on 'body'
        changed = False
        deleted_here = []

        # Split body into semi-colon terminated units but keep them for safe rebuild.
        units = []
        buf = []
        depth = 0
        for ch in body:
            buf.append(ch)
            if ch == '(':
                depth += 1
            elif ch == ')':
                depth -= 1 if depth > 0 else 0
            elif ch == ';' and depth == 0:
                units.append("".join(buf))
                buf = []
        # Remainder (blocks without ';', e.g., 'always' ... 'end') stays as-is
        tail = "".join(buf)

        rebuilt_units = []
        for u in units:
            parsed = parse_instance_statement(u)
            if parsed is None:
                rebuilt_units.append(u)  # untouched
                continue
            did_change, new_u, deleted = delete_instances_in_stmt(parsed, names, patterns)
            if did_change:
                changed = True
                deleted_here.extend(deleted)
            if new_u:
                rebuilt_units.append(new_u)

        # Reassemble
        new_body = "".join(rebuilt_units) + tail
        if changed:
            report["modules_edited"].append(mod_name)
            report["deleted_instances"][mod_name] = deleted_here
            global_deleted.update(deleted_here)
            out_parts.append(header + new_body + end_kw)
        else:
            out_parts.append(block)

        cursor = e

    # Copy trailing text after last module
    out_parts.append(verilog_text[cursor:])
    modified = "".join(out_parts)

    # Populate not_found (exact names only, patterns are not enumerable)
    if names:
        requested = set(names)
        report["not_found"] = sorted(list(requested - global_deleted))

    return modified, report

# ---------- Example usage ----------
if __name__ == "__main__":
    demo = r'''
module foo (input a,b, output y);
  wire n1, n2;
  MYCELL #(.WIDTH(8)) u0 (.A(a), .B(b), .Y(n1)),
                      inst_del (.A(a), .B(n1), .Y(n2));
  INV u_keep (.A(n2), .Y(y));
  INV u_del  (.A(n1), .Y(n2));
endmodule
'''
    keep, rep = remove_instances_from_verilog(
        demo,
        instance_names=["u_del"],
        instance_patterns=[r"^inst_del$"],  # regex
        limit_to_modules=None,              # or ["foo"]
    )
    print("=== Report ===")
    print(rep)
    print("\n=== After ===")
    print(keep)

    How to use in your project

    from remove_verilog_instances import remove_instances_from_verilog

with open("design_netlist.v", "r", encoding="utf-8") as f:
    net = f.read()

modified, report = remove_instances_from_verilog(
    net,
    instance_names=["U1234", "spare_inv_0"],
    instance_patterns=[r"^spare_.*$", r"^DBG_"],  # regex patterns
    limit_to_modules=["top", "cpu_core"]          # or None for all modules
)

# Validation (recommended)
if not report["modules_parsed"]:
    raise RuntimeError("No modules parsed; check input.")
# Optional: verify that requested names are gone
for nm in ["U1234", "spare_inv_0"]:
    assert nm not in modified, f"{nm} still present after deletion!"

with open("design_netlist.cleaned.v", "w", encoding="utf-8") as f:
    f.write(modified)

print("Edited modules:", report["modules_edited"])
print("Deleted instances:", report["deleted_instances"])
print("Not found:", report["not_found"])

    Notes & limitations (documented)

    • Targets instance deletion only (no insert/modify).

    • Handles common forms: CELL #(params) instA(...), instB(...); and single-instance statements.

    • Basic parser (regex + small state machine). No preprocessor/include, generate-conditionals, UDP, or exotic syntax support.

    • Comment stripping is simple; string-literal corner cases are not exhaustively handled (rare in flat netlists).

    If you want, I can adapt the matcher for fnmatch-style globs (*, ?) or add a tiny CLI (argparse) to run this from the command line with --names/--patterns/--modules.
